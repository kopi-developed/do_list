# Creates a desktop shortcut "do_list" that runs open.bat. ASCII-only on purpose.
$ErrorActionPreference = 'Stop'
$app  = Split-Path -Parent $MyInvocation.MyCommand.Path
$root = Split-Path -Parent $app
$desktop = [Environment]::GetFolderPath('Desktop')
$lnk = Join-Path $desktop 'do_list.lnk'
$sh = New-Object -ComObject WScript.Shell
$s = $sh.CreateShortcut($lnk)
$s.TargetPath = Join-Path $root 'open.bat'
$s.WorkingDirectory = $root
$s.WindowStyle = 7
$ico = Join-Path $app 'icon.ico'
if (Test-Path $ico) { $s.IconLocation = "$ico,0" }
else { $s.IconLocation = "$env:SystemRoot\System32\shell32.dll,269" }
$s.Save()
Write-Output "Shortcut created: $lnk"
